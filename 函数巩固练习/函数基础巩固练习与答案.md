### 题目 2：计算平均值

编写 `calculate_average(numbers)`，返回列表的平均值。

要求：列表为空时返回 `0`。

### 答案

```python
def calculate_average(numbers):
    if not numbers:
        return 0
    return sum(numbers) / len(numbers)


print(calculate_average([80, 90, 70]))  # 80.0
print(calculate_average([]))            # 0
```

### 题目 3：找出最大值

编写 `get_max(numbers)`，返回列表中的最大值。

要求：不能直接调用 `max()`；列表为空时返回 `None`。

### 答案

```python
def get_max(numbers):
    if not numbers:
        return None

    largest = numbers[0]
    for number in numbers[1:]:
        if number > largest:
            largest = number
    return largest


print(get_max([3, 8, 2, 10]))  # 10
print(get_max([]))              # None
```

### 题目 4：统计元音字母

编写 `count_vowels(text)`，统计字符串中 `a`、`e`、`i`、`o`、`u` 的数量，忽略大小写。

### 答案

```python
def count_vowels(text):
    vowels = "aeiou"
    count = 0

    for character in text.lower():
        if character in vowels:
            count += 1
    return count


print(count_vowels("Python Programming"))  # 4
```

### 题目 6：删除重复元素

编写 `remove_duplicates(items)`，返回删除重复元素后的列表，并保留原来的顺序。

### 答案

```python
def remove_duplicates(items):
    result = []

    for item in items:
        if item not in result:
            result.append(item)
    return result


print(remove_duplicates([1, 2, 1, 3, 2, 4]))  # [1, 2, 3, 4]
```

编写 `get_grade(score)`：

- 90～100：`A`
- 80～89：`B`
- 70～79：`C`

- 60 以下：`F`

```python
def get_grade(score):
    if score < 0 or score > 100:
        return "成绩无效"
    if score >= 90:
        return "A"
    if score >= 80:
        return "B"
    if score >= 70:
        return "C"
    if score >= 60:
        return "D"
    return "F"


print(get_grade(95))   # A
print(get_grade(75))   # C
print(get_grade(120))  # 成绩无效
```

### 题目 8：统计单词出现次数

编写 `count_words(text)`，统计一句英文中每个单词出现的次数，忽略大小写。

### 答案

```python
def count_words(text):
    words = text.lower().split()
    result = {}

    for word in words:
        result[word] = result.get(word, 0) + 1
    return result


text = "Python is easy and Python is useful"
print(count_words(text))
# {'python': 2, 'is': 2, 'easy': 1, 'and': 1, 'useful': 1}
```

### 题目 9：处理学生成绩

有如下数据：

```python
students = {
    "小明": [80, 90, 85],
    "小红": [95, 88, 92],
    "小刚": [60, 70, 65],
}
```

编写函数，完成以下功能：

1. 查询某个学生的平均分。
2. 找出平均分最高的学生。

### 答案

```python
def student_average(students, name):
    scores = students.get(name)
    if scores is None:
        return None
    return sum(scores) / len(scores)


def best_student(students):
    if not students:
        return None

    best_name = None
    best_score = -1

    for name in students:
        average = student_average(students, name)
        if average > best_score:
            best_name = name
            best_score = average

    return best_name, best_score


students = {
    "小明": [80, 90, 85],
    "小红": [95, 88, 92],
    "小刚": [60, 70, 65],
}

print(student_average(students, "小明"))
print(best_student(students))
```

### 题目 13：避免可变默认参数

下面的写法有什么问题？请改正：

```python
def add_item(item, items=[]):
    items.append(item)
    return items
```

### 答案

默认列表只会创建一次，多次调用会共用同一个列表。应使用 `None`：

```python
def add_item(item, items=None):
    if items is None:
        items = []

    items.append(item)
    return items


print(add_item("苹果"))  # ['苹果']
print(add_item("香蕉"))  # ['香蕉']
```

## 六、函数拆分练习：购物车

### 题目 14

编写一个购物车程序，支持：

1. 添加商品。
2. 删除商品。
3. 计算总价。
4. 显示购物车。

商品格式如下：

```python
cart = [
    {"name": "苹果", "price": 5, "count": 2},
    {"name": "牛奶", "price": 8, "count": 1},
]
```

### 答案

```python
def add_product(cart, name, price, count=1):
    product = {"name": name, "price": price, "count": count}
    cart.append(product)


def remove_product(cart, name):
    for product in cart:
        if product["name"] == name:
            cart.remove(product)
            return True
    return False


def calculate_total(cart):
    total = 0
    for product in cart:
        total += product["price"] * product["count"]
    return total


def show_cart(cart):
    for product in cart:
        print(
            f"{product['name']}：{product['price']}元 x "
            f"{product['count']}"
        )
    print(f"总价：{calculate_total(cart)} 元")


cart = []
add_product(cart, "苹果", 5, 2)
add_product(cart, "牛奶", 8)
show_cart(cart)
remove_product(cart, "苹果")
show_cart(cart)
```

## 七、综合小项目：学生成绩管理系统

### 题目 15

使用字典和函数完成一个菜单程序，支持：

```text
1. 添加学生
2. 删除学生
3. 查询学生
4. 修改成绩
5. 显示全部学生
6. 退出程序
```

### 参考答案

```python
def add_student(students):
    name = input("请输入学生姓名：").strip()
    if not name:
        print("姓名不能为空")
        return

    if name in students:
        print("学生已经存在")
        return

    try:
        score = float(input("请输入成绩："))
    except ValueError:
        print("成绩必须是数字")
        return

    if not 0 <= score <= 100:
        print("成绩必须在 0 到 100 之间")
        return

    students[name] = score
    print("添加成功")


def delete_student(students):
    name = input("请输入要删除的姓名：").strip()
    if name in students:
        del students[name]
        print("删除成功")
    else:
        print("学生不存在")


def find_student(students):
    name = input("请输入要查询的姓名：").strip()
    score = students.get(name)
    if score is None:
        print("学生不存在")
    else:
        print(f"{name} 的成绩是：{score}")


def update_score(students):
    name = input("请输入学生姓名：").strip()
    if name not in students:
        print("学生不存在")
        return

    try:
        score = float(input("请输入新成绩："))
    except ValueError:
        print("成绩必须是数字")
        return

    if 0 <= score <= 100:
        students[name] = score
        print("修改成功")
    else:
        print("成绩必须在 0 到 100 之间")


def show_all(students):
    if not students:
        print("暂无学生")
        return

    for name, score in students.items():
        print(f"{name}：{score}分")


def main():
    students = {}

    while True:
        print("\n--- 学生成绩管理系统 ---")
        print("1. 添加学生")
        print("2. 删除学生")
        print("3. 查询学生")
        print("4. 修改成绩")
        print("5. 显示全部学生")
        print("6. 退出程序")

        choice = input("请选择功能：").strip()

        if choice == "1":
            add_student(students)
        elif choice == "2":
            delete_student(students)
        elif choice == "3":
            find_student(students)
        elif choice == "4":
            update_score(students)
        elif choice == "5":
            show_all(students)
        elif choice == "6":
            print("程序结束")
            break
        else:
            print("请输入 1 到 6 之间的数字")


if __name__ == "__main__":
    main()
```
